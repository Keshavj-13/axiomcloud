"""
User Authentication & Management
Lightweight user tracking (Firebase handles actual auth)
"""
import logging
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from datetime import datetime
from app.core.database import get_db
from app.core.database import Base, engine
from sqlalchemy import Column, String, DateTime, Integer

logger = logging.getLogger(__name__)
router = APIRouter()


# Create users table if it doesn't exist
class User(Base):
    __tablename__ = "users"
    
    id = Column(Integer, primary_key=True, index=True)
    firebase_uid = Column(String(255), unique=True, index=True, nullable=False)
    email = Column(String(255), unique=True, index=True, nullable=False)
    display_name = Column(String(255), nullable=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    last_login = Column(DateTime, default=datetime.utcnow)


@router.post("/auth/user")
def register_or_update_user(
    firebase_uid: str,
    email: str,
    display_name: str = None,
    db: Session = Depends(get_db)
):
    """
    Called by frontend after successful Firebase auth.
    Creates or updates user record for tracking.
    """
    try:
        user = db.query(User).filter(User.firebase_uid == firebase_uid).first()
        
        if user:
            # Update last_login
            user.last_login = datetime.utcnow()
            if display_name:
                user.display_name = display_name
            db.commit()
            return {"id": user.id, "email": user.email, "status": "updated"}
        else:
            # Create new user
            user = User(
                firebase_uid=firebase_uid,
                email=email,
                display_name=display_name
            )
            db.add(user)
            db.commit()
            db.refresh(user)
            return {"id": user.id, "email": user.email, "status": "created"}
    except Exception as e:
        logger.error(f"Error registering user: {e}")
        raise HTTPException(status_code=500, detail="Failed to register user")


@router.get("/auth/user/{firebase_uid}")
def get_user(firebase_uid: str, db: Session = Depends(get_db)):
    """Get user information by Firebase UID."""
    try:
        user = db.query(User).filter(User.firebase_uid == firebase_uid).first()
        if not user:
            raise HTTPException(status_code=404, detail="User not found")
        return {
            "id": user.id,
            "email": user.email,
            "display_name": user.display_name,
            "created_at": user.created_at,
            "last_login": user.last_login
        }
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"Error fetching user: {e}")
        raise HTTPException(status_code=500, detail="Failed to fetch user")

Failure Cases

  • Missing dependency/import path errors.
  • Type/shape mismatch and invalid input values.
  • Null/undefined/None access from unchecked assumptions.
  • External IO failures (network, disk, database, auth).
  • Unhandled exceptions if error branches are incomplete.

Design Decisions

  • Modular composition to keep concerns separated.
  • Abstraction through functions/classes for readability and reuse.
  • Error-handling branches for resilience.
  • Configuration/constants for predictable behavior.
  • Trade-off: maintainability often prioritized over micro-optimization.

Performance Notes

  • Complexity grows with loop nesting, data size, and expensive calls.
  • Bottlenecks often come from repeated parsing or blocking IO.
  • Possible improvements: caching, batching, memoization, lazy loading, better algorithms.

Security Notes

  • Validate untrusted input and reject malformed payloads early.
  • Avoid leaking secrets in logs, stack traces, and client-visible output.
  • Use least-privilege credentials and secure defaults (auth, CORS, HTTPS).
  • Prefer explicit error handling that hides internals from attackers.

Common Beginner Questions

What is a file, and why split programs into many files?

A file stores text instructions. Splitting programs into files helps humans focus on one concern at a time.

Why do punctuation symbols matter so much?

Symbols like (), {}, [], ,, :, and quotes define syntax. Tiny punctuation changes can completely change meaning.

How do I know which line runs first?

Top-level code runs from top to bottom, then framework/callers invoke functions later when needed.
